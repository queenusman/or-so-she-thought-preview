Interactive 3D Touch & Mouse Reaction for the Book const bookContainer = document.querySelector('.book-container'); if (bookContainer) { const resetTilt = () => { bookContainer.style.transform = 'rotateY(-18deg) rotateX(6deg) translateY(0px)'; }; const handleMove = (clientX, clientY) => { const rect = bookContainer.getBoundingClientRect(); const x = clientX - rect.left; const y = clientY - rect.top; const centerX = rect.width / 2; const centerY = rect.height / 2; // Calculate tilt angles based on touch / cursor position const tiltY = ((x - centerX) / centerX) * 22; // left-to-right tilt const tiltX = -((y - centerY) / centerY) * 16; // up-and-down tilt bookContainer.style.transform = `rotateY(${tiltY}deg) rotateX(${tiltX}deg) translateY(-8px) scale(1.02)`; }; // Mouse Listeners (Desktop) bookContainer.addEventListener('mousemove', (e) => { handleMove(e.clientX, e.clientY); }); bookContainer.addEventListener('mouseleave', resetTilt); // Touch Listeners (Mobile & Tablets) bookContainer.addEventListener('touchstart', (e) => { const touch = e.touches[0]; handleMove(touch.clientX, touch.clientY); }, { passive: true }); bookContainer.addEventListener('touchmove', (e) => { const touch = e.touches[0]; handleMove(touch.clientX, touch.clientY); }, { passive: true }); bookContainer.addEventListener('touchend', resetTilt); bookContainer.addEventListener('touchcancel', resetTilt); }
